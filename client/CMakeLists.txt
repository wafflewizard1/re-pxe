cmake_minimum_required(VERSION 3.20)
project(repxe_client VERSION 0.13.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/media/Collect-BatteryHealth.ps1" BATTERY_HEALTH_SCRIPT)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS media/Collect-BatteryHealth.ps1)
configure_file(src/battery_health_script.hpp.in generated/battery_health_script.hpp @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/media/Cleanup-InstalledClient.ps1" CLEANUP_SCRIPT)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS media/Cleanup-InstalledClient.ps1)
configure_file(src/cleanup_script.hpp.in generated/cleanup_script.hpp @ONLY)
add_library(repxe_core src/core.cpp src/finish.cpp src/hardware.cpp src/battery_health.cpp src/install_progress.cpp third_party/pugixml.cpp)
target_include_directories(repxe_core PUBLIC include third_party)
target_include_directories(repxe_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
  target_compile_options(repxe_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(repxe_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(repxe_core_tests tests/core_tests.cpp)
target_link_libraries(repxe_core_tests PRIVATE repxe_core)
enable_testing()
add_test(NAME core COMMAND repxe_core_tests)
add_executable(repxe_hardware_tests tests/hardware_tests.cpp)
target_link_libraries(repxe_hardware_tests PRIVATE repxe_core)
add_test(NAME hardware COMMAND repxe_hardware_tests)
add_executable(repxe_battery_tests tests/battery_tests.cpp)
target_link_libraries(repxe_battery_tests PRIVATE repxe_core)
add_test(NAME battery COMMAND repxe_battery_tests)
add_executable(repxe_progress_tests tests/install_progress_tests.cpp)
target_link_libraries(repxe_progress_tests PRIVATE repxe_core)
add_test(NAME install_progress COMMAND repxe_progress_tests)
add_executable(repxe_handoff_tests tests/handoff_tests.cpp)
target_link_libraries(repxe_handoff_tests PRIVATE repxe_core)
add_test(NAME customer_handoff COMMAND repxe_handoff_tests)
if(WIN32)
  add_library(repxe_windows_trust src/windows_trust.cpp)
  target_include_directories(repxe_windows_trust PUBLIC include)
  target_compile_definitions(repxe_windows_trust PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A000000)
  target_link_libraries(repxe_windows_trust PUBLIC wintrust crypt32 version)
  add_executable(repxe_windows_trust_tests tests/windows_trust_tests.cpp)
  target_link_libraries(repxe_windows_trust_tests PRIVATE repxe_windows_trust)
  add_test(NAME windows_trust COMMAND repxe_windows_trust_tests)
  # Keep Windows headers from replacing standard C++ min/max calls.
  target_compile_definitions(repxe_core PUBLIC NOMINMAX)
  add_executable(repxe_wua_tests tests/windows_update_tests.cpp src/windows_update.cpp)
  target_compile_definitions(repxe_wua_tests PRIVATE REPXE_WUA_TESTS UNICODE _UNICODE)
  target_link_libraries(repxe_wua_tests PRIVATE repxe_core ole32 oleaut32 uuid)
  add_executable(repxe_operation_lock_tests tests/operation_lock_tests.cpp)
  target_link_libraries(repxe_operation_lock_tests PRIVATE repxe_core)
  add_test(NAME operation_lock COMMAND repxe_operation_lock_tests)
  add_executable(repxe_state_file_tests tests/state_file_tests.cpp)
  target_link_libraries(repxe_state_file_tests PRIVATE repxe_core ole32)
  add_test(NAME state_file COMMAND repxe_state_file_tests)
  add_executable(re-pxe src/main.cpp src/windows.cpp src/windows_update.cpp src/technician_network.cpp src/technician_ui.cpp src/pxe_ui.cpp src/driver_console.cpp)
  target_link_libraries(re-pxe PRIVATE repxe_core repxe_windows_trust bcrypt setupapi advapi32 ole32 oleaut32 uuid cfgmgr32 crypt32 shell32 winhttp user32 gdi32 comdlg32 mpr)
  target_compile_definitions(re-pxe PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  if(MINGW)
    target_link_options(repxe_windows_trust_tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(repxe_handoff_tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(re-pxe PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(repxe_core_tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(repxe_progress_tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(repxe_hardware_tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(repxe_battery_tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(repxe_operation_lock_tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(repxe_state_file_tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(repxe_wua_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
